package "Симулятор полёта ракеты" {
  [Пользовательский интерфейс\n(main.cpp)] as UI
  [Менеджер объектов\n(ObjectManager)] as ObjMgr
  [Модель мира\n(SimpleWorld)] as World
  [Решатель ОДУ\n(RungeKutta4Solver)] as Solver
  [Система динамики\n(FullRocketODE)] as ODE
  [Ракета\n(MANPAD_V1 / AbstractAircraft)] as Rocket
  [Провайдер параметров\n(DynamicParametersProvider)] as ParamProv
  [Интерполятор данных\n(ComponentInterpolationManager)] as InterpMgr
  [Аэродинамическая модель\n(IAeroModel / FullAeroModelAdapter)] as Aero
  [Сохранение результатов\n(SimulationMomento + CsvSaveStrategy)] as Saver
}

UI --> ObjMgr : управляет жизненным циклом
UI --> World : конфигурирует окружение
UI --> InterpMgr : загружает таблицы (тяга, масса и т.д.)
UI --> ParamProv : создаёт провайдер
UI --> Rocket : создаёт экземпляр
UI --> Saver : настраивает сохранение

ObjMgr --> Rocket : владеет и отслеживает
ObjMgr --> Solver : передаёт объекты для интеграции

Solver --> ODE : вызывает get_rhs_derivatives()
Solver --> Saver : передаёт снимки состояния

ODE --> ParamProv : получает массу, инерцию, тягу
ODE --> World : запрашивает плотность, гравитацию и т.д.
ODE --> Aero : вычисляет аэродинамические силы/моменты

ParamProv --> InterpMgr : делегирует интерполяцию
ParamProv --> Aero : предоставляет аэродинамическую модель

Rocket --> ODE : содержит систему динамики
Rocket --> Aero : использует аэродинамику
Rocket --> ParamProv : наблюдает через weak_ptr

Saver --> UI : сохраняет CSV по завершении

note right of Rocket
  Абстрактный компонент,
  объединяющий все типы ЛА:
  MANPAD_V1, GuidedMissile и др.
end note

note right of InterpMgr
  Загружает внешние данные:
  - Тяга(t)
  - Масса(t)
  - Инерция(t)
  - Аэродинамические коэффициенты(α, M)
end note

@enduml
